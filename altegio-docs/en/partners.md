(function initCarousel() {
  var radios = document.querySelectorAll('.vcarousel__radio');
  if (!radios.length) { setTimeout(initCarousel, 500); return; }

  var current = 0;
  var total = radios.length;
  var INTERVAL = 4000;

  function advance() {
    current = (current + 1) % total;
    radios[current].checked = true;
  }

  var timer = setInterval(advance, INTERVAL);

  document.querySelectorAll('.vcarousel__dot').forEach(function(dot, i) {
    dot.addEventListener('click', function() {
      current = i;
      clearInterval(timer);
      timer = setInterval(advance, INTERVAL);
    });
  });
})();
